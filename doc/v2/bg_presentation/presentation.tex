\section{Мотивация}
% ══════════════════════════════════════════════════════════

\begin{frame}{Проблемът със заявките, проверявани при изпълнение}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \textbf{Традиционен подход:}
            \begin{itemize}\small
                \item Заявките се описват като \alert{низове}
                \item Валидиране едва при изпълнение
                \item Грешни имена на полета → срив при изпълнение
                \item Несъвместими типове → късно откриване
                \item Смяна на хранилището → \alert{пълно пренаписване}
            \end{itemize}
            \vspace{0.15em}
            \textbf{Изследователски въпрос:}\\
            \textit{Може ли компилаторът да поеме ролята на
            първи валидатор на заявките?}
        \end{column}
        \begin{column}{0.43\textwidth}
            \small
            \begin{tabularx}{\linewidth}{@{}XX@{}}
                \toprule
                \textbf{При изпълнение} & \textbf{При компилация}\\
                \midrule
                Низови заявки &
                    \texttt{constexpr} междинно представяне\\
                {Грешка при\newline изпълнение} & {Грешка при\newline \alert{компилация}}\\
                Късно откриване &
                    Ранно откриване\\
                \bottomrule
            \end{tabularx}
            \vspace{0.15em}
            \begin{alertblock}{Цел}
                {\small Компилаторът поема ролята
                на първи валидатор
                на заявките.}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

                \node[layerbox, below=of ir] (conn)
                    {\textbf{Connector слой}\\
                    {\scriptsize\texttt{connector\_trait<DB>}}};
                \node[layerbox, fill=gray!10, draw=gray!50,
                    below=of conn] (drv)
                    {\textbf{Driver / runtime}\\
                    SQL, BSON, Cypher, CQL};
                \draw[ormline] (app)  -- (ent);
                \draw[ormline] (ent)  -- (ir);
                \draw[ormline] (ir)   -- (conn);
                \draw[ormline] (conn) -- (drv);
            \end{tikzpicture}
        \end{column}
        \begin{column}{0.55\textwidth}
            \textbf{Ключови архитектурни принципа:}
            \begin{itemize}\footnotesize
                \setlength{\itemsep}{0.15em}
                \item Без виртуален интерфейс --- признаков (trait-based) договор
                \item \texttt{constexpr} междинно представяне на заявката --- ранна валидация
                \item Маркери за способности (capability tags) ограничават неподдържаните операции
                \item Смяна на системата за съхранение = смяна само на типа
            \end{itemize}
            \begin{alertblock}{Централен инвариант}
                {\footnotesize Приложният код не комуникира директно с драйвера --- работи само с C++ типове.}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\section{Реализация}
% ══════════════════════════════════════════════════════════

\begin{frame}[fragile]{Минимален compile-time пример}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
\begin{lstlisting}
struct User {
  orm::property<int, "id"> id;
  orm::property<std::string, "name"> name;
};

static constexpr auto q =
  orm::select(
    orm::field<&User::id>,
    orm::field<&User::name>)
  .where(orm::field<&User::id>
    == orm::Placeholder<int>{});
\end{lstlisting}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{itemize}\small
                \item \texttt{property<T,"name">} носи тип и compile-time име на колона
                \item Заявката се изгражда като \texttt{constexpr} IR, а не като низ
                \item Грешно поле или неподдържана операция се открива рано
            \end{itemize}
            \vspace{0.2em}
            \begin{alertblock}{Ключова идея}
                {\small Компилаторът поема ролята на първи валидатор на заявката.}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Договор на конектора и механизъм на способностите}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \textbf{Минимален договор на конектора:}
\begin{lstlisting}
template<>
struct orm::connector_trait<MyDB> {
  template<typename T> struct wire_type;
  using cursor_type = ...;

  using supports_transactions = true_type;

  static auto execute(
    MyDB& db, auto query,
    auto&&... params);
};
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{itemize}\small
                \item \texttt{is\_connector<DB>} валидира интерфейса при компилация
                \item Маркерите за способности ограничават неподдържаните операции
                \item Смяната на системата за съхранение е смяна на типа, а не на интерфейса за заявки
            \end{itemize}
            \vspace{0.15em}
            \begin{block}{Демонстрирани системи за съхранение}
                {\small SQLite, PostgreSQL, MySQL, MongoDB, Redis, Neo4j, Cassandra}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Един модел, много системи за съхранение}
    \begin{columns}[T]

            {\small Едно междинно представяне → различни системи:}
            \begin{tabularx}{\linewidth}{@{}lX@{}}
                \toprule
                \textbf{Система}  & \textbf{Материализация}\\
                \midrule
                SQLite      & \texttt{WHERE id = ?}\\
                PostgreSQL  & \texttt{WHERE id = \$1}\\
                MySQL       & \texttt{WHERE id = ?}\\
                \midrule
                MongoDB     & BSON filter\\
                Redis       & \texttt{GET}/\texttt{SET}/\texttt{DEL}\\
                Neo4j       & Cypher \texttt{MATCH…WHERE}\\
                Cassandra   & CQL \texttt{SELECT…FROM}\\
                \bottomrule
            \end{tabularx}
        \end{column}
        \begin{column}{0.47\textwidth}
            \textbf{Абстракция на връзките:}\\[0.1em]
            \begin{tabularx}{\linewidth}{@{}lX@{}}
                \texttt{embed}     & MongoDB вложен документ\\
                \texttt{reference} & SQL foreign key / JOIN\\
                                   & Redis: key prefix\\
                                   & Neo4j: edge\\
            \end{tabularx}
            \vspace{0.1em}
            \begin{alertblock}{Принцип}
                {\small C++ моделът е единственият
                \emph{логически} носител на схемата.
                Системата за съхранение определя физическата
                материализация.}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\section{Приноси и заключение}

\begin{frame}{Основни резултати и извод}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \begin{enumerate}\small
                \item ORM модел с проверка при компилация в стандартен C++23, без външна кодогенерация
                \item Един C++ обектен модел за 7 системи за съхранение
                \item Формален договор на конектора и разширяемост чрез способности
                \item Асинхронен слой с измерима полза при натоварвания, ограничени от вход-изход
                \item Многостепенна верификация: 271 теста + проверки срещу живи системи + емпирични измервания
            \end{enumerate}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Основен извод}
                {\footnotesize Освен ORM с проверка при изпълнение (удобство, но късно откриване на грешки) и прекия достъп през драйвера (контрол, но без абстракция), е възможна \textbf{трета алтернатива}: договор на конектора, валидиран от компилатора, плюс корутинно изпълнение.}
            \end{block}
            \textbf{Бъдещо развитие:}
            \begin{itemize}\footnotesize
                \setlength{\itemsep}{0.1em}
                \item Отражение (reflection) от C++26
                \item Измервания срещу мрежови сървъри
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
