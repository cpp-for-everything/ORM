\section{Реализация}
% ══════════════════════════════════════════════════════════

\begin{frame}[fragile]{Минимален compile-time пример}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
\begin{lstlisting}
struct User {
  orm::property<int, "id"> id;
  orm::property<std::string, "name"> name;
};

static constexpr auto q =
  orm::select(
    orm::field<&User::id>,
    orm::field<&User::name>)
  .where(orm::field<&User::id>
    == orm::Placeholder<int>{});
\end{lstlisting}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{itemize}\small
                \item \texttt{property<T,"name">} носи тип и compile-time име на колона
                \item Заявката се изгражда като \texttt{constexpr} IR, а не като низ
                \item Грешно поле или неподдържана операция се открива рано
            \end{itemize}
            \vspace{0.2em}
            \begin{alertblock}{Ключова идея}
                {\small Компилаторът поема ролята на първи валидатор на заявката.}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Договор на конектора и механизъм на способностите}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \textbf{Минимален договор на конектора:}
\begin{lstlisting}
template<>
struct orm::connector_trait<MyDB> {
  template<typename T> struct wire_type;
  using cursor_type = ...;

  using supports_transactions = true_type;

  static auto execute(
    MyDB& db, auto query,
    auto&&... params);
};
\end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{itemize}\small
                \item \texttt{is\_connector<DB>} валидира интерфейса при компилация
                \item Маркерите за способности ограничават неподдържаните операции
                \item Смяната на системата за съхранение е смяна на типа, а не на интерфейса за заявки
            \end{itemize}
            \vspace{0.15em}
            \begin{block}{Демонстрирани системи за съхранение}
                {\small SQLite, PostgreSQL, MySQL, MongoDB, Redis, Neo4j, Cassandra}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Приноси и заключение}

\begin{frame}{Основни резултати и извод}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \begin{enumerate}\footnotesize
                \setlength{\itemsep}{0.15em}
                \item ORM модел с проверка при компилация в стандартен C++23, без външна кодогенерация
                \item Един C++ обектен модел за 7 системи за съхранение (SQL и NoSQL)
                \item \textbf{Релационно-осъзнати заявки}: авто-JOIN, изведен при компилация; една заявка $\rightarrow$ SQL \texttt{JOIN} / Mongo \texttt{\$lookup}, жив \texttt{joined\_row} от 4 бази
                \item Формален договор на конектора и разширяемост чрез способности
                \item Асинхронен слой с измерима полза при натоварвания, ограничени от вход-изход
                \item Многостепенна верификация: тестове $+$ проверки срещу живи системи $+$ емпирични измервания
            \end{enumerate}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Основен извод}
                {\footnotesize \textbf{Трета алтернатива} освен ORM при изпълнение (късно
                откриване) и пряк достъп през драйвера (без абстракция): договор на
                конектора, валидиран от компилатора, плюс корутинно изпълнение.
                \textbf{\color{tusofia}Тази комбинация не съществува в друга C++ библиотека.}}
            \end{block}
            {\footnotesize Един и същ код --- \textbf{macOS / Linux / Windows}, релационни и
            документни бази; демонстрациите вървят „out of the box" през Docker.}
            \textbf{Бъдещо развитие:}
            \begin{itemize}\footnotesize
                \setlength{\itemsep}{0.1em}
                \item Отражение (reflection) от C++26
                \item Измервания срещу мрежови сървъри
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

    {\footnotesize Седем демонстрации: шест с предварително \emph{уловен} изход (без жива
    компилация на сцената) $+$ едно \emph{живо} multi-store изпълнение през Docker.}
    \vspace{0.3em}
    \footnotesize
    \renewcommand{\arraystretch}{1.3}
    \begin{tabularx}{\linewidth}{@{}p{1.3cm}X@{}}
        \toprule
        & \textbf{Демонстрира}\\
        \midrule
        Демо 1 & MySQL приема, MongoDB отказва JOIN\\
        Демо 2 & Заявката е \texttt{constexpr} стойност, не низ\\
        Демо 3 & Несъвместим тип поле\,↔\,стойност\\
        Демо 4 & Грешен брой/тип на параметрите\\
        Демо 5 & Redis: само по първичен ключ\\
        Демо 6 & relationship движи заявката (авто-JOIN, MockDB)\\

% ── Демо 1 (compile-time) ──────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 1 — Един модел: MySQL приема, MongoDB отказва}
    {\footnotesize Една и съща SELECT\,…\,JOIN заявка срещу два конектора:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
auto q = select(field<&User::id>, field<&Post::body>)
   .join<inner, Post>(field<&User::id> == field<&Post::author_id>);

db<MySQLDB>{conn} << q;   // relational -> compiles
db<MongoDB>{conn} << q;   // document   -> REJECTED
\end{lstlisting}
    \lstinputlisting[basicstyle=\scriptsize\ttfamily,breaklines=true,keywordstyle=\color{black},stringstyle=\color{black},commentstyle=\color{cppgray}]{demo/01_capability_gating/compiler-output-slide.txt}
\end{frame}

% ── Демо 2 (positive) ──────────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 2 — Заявката е constexpr стойност, не низ}
    {\footnotesize Структурата е в типа и се проверява при компилация --- нищо не се
    изпълнява.}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
constexpr auto q = select(field<&User::id>, field<&User::name>)
                      .where(field<&User::score> > 0.0);

static_assert(decltype(q)::response_type::size == 2);
static_assert(decltype(q.where_clauses())::size == 1);
\end{lstlisting}
    {\scriptsize Имена на колони, проекция и клаузи --- проверени без изпълнение. (SOCI гради низ при изпълнение.)}
\end{frame}

% ── Демо 3 (compile-time) ──────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 3 — Несъвместим тип поле↔стойност}
    {\footnotesize Операторите сверяват C++ типа на колоната и на стойността:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
field<&User::score> > 0;                     // double vs int -> OK
field<&User::id>    == std::u8string{u8"x"};  // int vs string -> REJECTED
\end{lstlisting}
    \lstinputlisting[basicstyle=\scriptsize\ttfamily,breaklines=true,keywordstyle=\color{black},stringstyle=\color{black},commentstyle=\color{cppgray}]{demo/03_type_safety/compiler-output-slide.txt}
\end{frame}

% ── Демо 4 (compile-time) ──────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 4 — Брой и тип на параметрите}
    {\footnotesize Заявка с два анонимни placeholder-а, извикана с грешен брой аргументи:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
auto q = select(field<&User::id>)
   .where( (field<&User::id>    == Placeholder<int>{}) &&
           (field<&User::score> >  Placeholder<double>{}) );

db.execute(q, 7, 3.5);   // 2 args -> OK
db.execute(q, 7);        // 1 arg  -> REJECTED
\end{lstlisting}
    \lstinputlisting[basicstyle=\scriptsize\ttfamily,breaklines=true,keywordstyle=\color{black},stringstyle=\color{black},commentstyle=\color{cppgray}]{demo/04_param_arity/compiler-output-slide.txt}
\end{frame}

% ── Демо 5 (compile-time) ──────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 5 — Redis: само по първичен ключ}
    {\footnotesize Заявка с не-PK предикат срещу Redis конектора:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
select(field<&Session::token>)
   .where(field<&Session::id> == Placeholder<int>{});       // PK -> OK

select(field<&Session::token>)
   .where(field<&Session::id>    == Placeholder<int>{})
   .where(field<&Session::token> == Placeholder<u8string>{}); // non-PK -> REJECTED
\end{lstlisting}
    \lstinputlisting[basicstyle=\scriptsize\ttfamily,breaklines=true,keywordstyle=\color{black},stringstyle=\color{black},commentstyle=\color{cppgray}]{demo/05_backend_shape/compiler-output-slide.txt}
\end{frame}

% ── Демо 6 (positive) ──────────────────────────────────────────────────────
\begin{frame}[fragile]{Демо 6 — relationship движи заявката (автоматичен JOIN)}
    {\footnotesize \texttt{relationship<reference<\&User::id>, "user\_id">} е истинска
    FK колона; избор на поле от свързана таблица извежда JOIN-а при компилация:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
struct Post {
  property<int,"id"> id;  property<u8string,"body"> body;
  relationship<store_as::reference<&User::id>, "user_id"> user_id;
};

db << select(field<&Post::id>, field<&User::name>);
\end{lstlisting}
    {\scriptsize $\rightarrow$ изведен и проверен при компилация SQL:}
\begin{lstlisting}[basicstyle=\scriptsize\ttfamily,keywordstyle=\color{black}]
SELECT posts.id, user.name
  FROM posts INNER JOIN user ON posts.user_id = user.id
\end{lstlisting}
    {\scriptsize \texttt{optional\_field} $\rightarrow$ LEFT / RIGHT / FULL (матрица 2$\times$2);
    многозвенно Comment\,$\rightarrow$\,Post\,$\rightarrow$\,User се извежда автоматично.}
\end{frame}

% ── Демо 7 (live, multi-store) ─────────────────────────────────────────────
\begin{frame}[fragile]{Демо 7 — Една заявка, четири живи бази (SQL\,$\times$\,NoSQL)}
    {\footnotesize Демо 6 показа \emph{рендерирането}; тук същата заявка се \emph{изпълнява}
    срещу живи бази през \texttt{docker compose} --- релационни \emph{и} документна:}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
constexpr auto q =
    select(field<&Book::title>, field<&Author::name>);
auto rows = (db << q).to_vector();  // vector<joined_row<Book,Author>>
\end{lstlisting}
    {\scriptsize $\rightarrow$ изпълнено срещу всяко хранилище (един и същ C++):}
\begin{lstlisting}[basicstyle=\scriptsize\ttfamily,breaklines=true,keywordstyle=\color{black},stringstyle=\color{black},commentstyle=\color{cppgray}]
SQLite / PostgreSQL / MySQL : INNER JOIN authors ON books.author_id = authors.id
MongoDB                     : $lookup { from: authors, localField: author_id,
                                        foreignField: id } + $unwind
all four -> joined_row{ "The Hobbit"->Tolkien,
                        "A Wizard of Earthsea"->Le Guin }
\end{lstlisting}
    {\scriptsize Един модел, една заявка $\rightarrow$ SQL JOIN \emph{и} \texttt{\$lookup},
    идентичен \texttt{joined\_row}. „Out of the box" през Docker --- еднакво на
    macOS / Linux / Windows.}
\end{frame}

\begin{frame}[fragile]{Какво още може да се провери при компилация}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Вече се проверява:}
            \begin{itemize}\footnotesize
                \setlength{\itemsep}{0.05em}
                \item Договор и способности на конектора
                \item Тип на заявката и видове клаузи
                \item Обвързване поле\,↔\,таблица\,↔\,колона
                \item Консистентност на проекцията
                \item Валидност на placeholder-ите
                \item Структурна форма за конкретния backend
                \item \textbf{Съвместимост поле\,↔\,стойност по тип}
                \item \textbf{Брой/тип на параметрите} (анонимни)
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Архитектурно възможно (надграждане):}
            \begin{itemize}\footnotesize
                \setlength{\itemsep}{0.05em}
                \item Аритет при \emph{индексирани} параметри \\ --- по дефиниция договор на конектора
                \item Compile-time валидиране на \texttt{enum\_t}/\texttt{set\_t}
                \item Референтна цялост на заявката
                \item Деривиране на схемата чрез C++26 reflection
            \end{itemize}
        \end{column}
    \end{columns}
    \vspace{0.05em}
    {\scriptsize Две скорошни проверки --- всяка е отказ при компилация:}
\begin{lstlisting}
field<&User::id> == std::u8string{u8"x"};  // incompatible type
db.execute(query_with_2_placeholders, 5);  // wrong arg count
\end{lstlisting}
\end{frame}

\begin{frame}{Грешки, уловени при компилация}
    \scriptsize
    \setlength{\tabcolsep}{4pt}
    \renewcommand{\arraystretch}{1.0}
    \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{6.9cm}>{\raggedright\arraybackslash}X@{}}
        \toprule
        \textbf{Клас грешка} & \textbf{Уловена при компилация чрез}\\
        \midrule
        Неподдържана операция за backend (JOIN срещу MongoDB)
            & \texttt{static\_assert} за липсваща \texttt{supports\_joins}\\
        Несъществуващо / грешно име на колона
            & поле = член-указател \texttt{\&T::m}; име от \texttt{property<T,"..">}\\
        Достъп до неселектирана колона
            & \texttt{static\_assert}: „field not found in projection"\\
        Малформирана заявка
            & concepts \texttt{is\_field} / \texttt{is\_rule}\\
        Несъвместим тип на сравнение (\texttt{int} колона vs \texttt{string})
            & \texttt{static\_assert}: „incompatible operand types"\\
        Невалиден индекс на placeholder
            & \texttt{requires std::is\_placeholder<...>}\\
        Нарушена форма за backend (Redis / Cassandra)
            & \texttt{consteval} + \texttt{static\_assert}\\
        Неспециализиран конектор
            & \texttt{static\_assert} в \texttt{connector\_trait<DB>}\\
        \bottomrule
    \end{tabularx}
    \vspace{0.15em}

    {\scriptsize\itshape Всяка от тези грешки в традиционните ORM библиотеки би била
    изключение или SQL грешка при изпълнение; тук е отказ на компилатора.}
\end{frame}
